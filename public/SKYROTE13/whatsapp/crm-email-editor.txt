function activityTime(activity) {
  const value = activity.occurred_at || (["contacto", "respuesta"].includes(activity.kind) ? activity.created_at : "") || (activity.date ? `${activity.date}T${activity.time || "12:00"}:00-05:00` : activity.created_at);

function contactHistory(companyId, data) {
  const rows = (data.activities || []).filter((row) => row.company_id === companyId);
  const attempts = rows.filter((row) => row.kind === "contacto" || ["correo", "whatsapp", "llamada", "reunion", "presencial"].includes(row.kind));
  attempts.sort((a, b) => activityTime(b).localeCompare(activityTime(a)));
  const contact = attempts[0];
  if (!contact) {
    const responses2 = rows.filter((row) => row.kind === "respuesta").sort((a, b) => activityTime(b).localeCompare(activityTime(a)));
    return { contact: null, at: "", channels: [], response: responses2[0] || null };
  }
  const at = activityTime(contact);
  const group = attempts.filter((row) => activityTime(row) === at);
  const channels = [...new Set(group.flatMap((row) => row.kind === "contacto" ? row.channels || [] : [row.kind === "reunion" ? "presencial" : row.kind]))].filter((key) => CONTACT_CHANNELS[key]);
  const responses = rows.filter((row) => row.kind === "respuesta" && (row.contact_id === contact.id || !row.contact_id && activityTime(row) >= at));
  responses.push(...group.filter((row) => ["conversacion", "no_interes", "no_contactar"].includes(row.outcome)));
  responses.sort((a, b) => activityTime(b).localeCompare(activityTime(a)));
  return { contact, at, channels, response: responses[0] || null };
}
function companyStage(company, data) {
  if (STAGES[company.pipeline_stage]) return company.pipeline_stage;
  if (["no_contactar", "no_encaja"].includes(company.status)) return "perdida";
  const deals = (data.deals || []).filter((row) => row.company_id === company.id && !row.archived);
  const open = deals.filter(isOpen);
  if (open.some((row) => dealStageKey(row) === "propuesta")) return "propuesta";
  if (!open.length && (deals.some((row) => row.stage === "ganada") || company.status === "cliente")) return "ganada";
  if (!open.length && deals.some((row) => row.stage === "perdida")) return "perdida";
  if (["conversacion", "calificada"].includes(company.status)) return "propuesta";
  const history = contactHistory(company.id, data);
  if (history.response?.outcome === "conversacion") return "propuesta";
  if (history.response && ["no_interes", "no_contactar"].includes(history.response.outcome)) return "perdida";
  if (history.contact || company.status === "contactada" || open.some((row) => dealStageKey(row) === "demo")) return "demo";
  return "diagnostico";
}

function stageOperations(data, company, stage, { reason = "", now = /* @__PURE__ */ new Date(), allowContact = false } = {}) {
  if (!STAGES[stage]) throw new Error("Elige una etapa v\xE1lida.");
  if (stage === "demo" && !allowContact) throw new Error("Registra los canales antes de pasar a Contactado.");
  if (stage === "ganada") throw new Error("Registra el importe y la fecha para confirmar una venta ganada.");
  if (stage === "perdida" && !reason.trim()) throw new Error("Indica el motivo del cierre.");
  const operations = [op("companies", companyUpdate(company, stage))];
  const open = data.deals.filter((deal) => deal.company_id === company.id && isOpen(deal));
  for (const deal of open) operations.push(op("deals", { ...deal, stage, pipeline_version: 2, closed_date: stage === "perdida" ? today(now) : "", lost_reason: stage === "perdida" ? reason.trim() : "", next_step: stage === "perdida" ? "" : deal.next_step || "", next_action_date: stage === "perdida" ? "" : deal.next_action_date || "" }));
  if (stage === "perdida") {
    for (const task of data.tasks.filter((task2) => task2.company_id === company.id && isPending(task2))) {
      const deal = data.deals.find((deal2) => deal2.id === task.deal_id);
      if (deal?.stage === "ganada" || task.kind === "entrega") continue;
      operations.push(op("tasks", { ...task, status: "cancelada", cancel_reason: reason.trim() }));

}
function contactOperations(data, company, channels, { body = "", move = false, allowReopen = false, now = /* @__PURE__ */ new Date() } = {}) {
  if (!Array.isArray(channels) || channels.some((key) => !CONTACT_CHANNELS[key])) throw new Error("Revisa los canales seleccionados.");
  const chosen = Object.keys(CONTACT_CHANNELS).filter((key) => channels.includes(key));
  if (!chosen.length) throw new Error("Elige al menos un canal de contacto.");
  const current = companyStage(company, data);
  if (current === "perdida" && !allowReopen) throw new Error("El seguimiento est\xE1 cerrado. Reabre la empresa antes de contactarla.");
  const stage = move || current === "diagnostico" ? "demo" : current;
  let operations;
  if (stage !== current || move) operations = stageOperations(data, company, stage, { allowContact: true, now });
  else operations = [op("companies", companyUpdate(company, stage))];
  const contact = { id: id(), company_id: company.id, deal_id: "", kind: "contacto", channels: chosen, outcome: "sin_respuesta", body: body.trim() || `Contacto realizado por ${chosen.map((key) => CONTACT_CHANNELS[key]).join(" y ")}.`, ...stamp(now) };
  operations.push(op("activities", contact));
  return assertSize(operations);
}

function normalizeWhatsAppPhone(value) {
  let digits = String(value || "").replace(/\D/g, "").replace(/^00/, "");
  if (/^9\d{8}$/.test(digits)) digits = "51" + digits;
  return /^[1-9]\d{7,14}$/.test(digits) ? "+" + digits : "";
}

    return this.dispatch(job, input.dryRun);
  }
  async dispatch(job, dryRun, recover = false) {
    if (["queued", "processing", "sent", "cancelled", "failed"].includes(job.status) || job.status === "uncertain" && !recover) return publicJob(await this.refreshInternal(job));
    if (["submitting", "validating"].includes(job.status)) return publicJob(await this.refreshInternal(job));
    if (!this.configured) fail("LR_NOT_CONFIGURED", "Falta configurar LR.", 503);
    if (!dryRun && !this.enabled) fail("SENDING_DISABLED", "El env\xEDo real todav\xEDa no est\xE1 activado.", 503);
    if (["RATE_LIMITED", "QUEUE_LIMIT"].includes(job.error_code) && Date.parse(job.lease_until) > this.now().getTime()) fail("RATE_LIMITED", "LR pidi\xF3 esperar antes de volver a intentar.", 429);

    job = await this.store.claimJob(job, dryRun);
    let mayHaveSent = false;
    try {
      if (!recover && !dryRun) {
        const connection = await this.provider.connection();
        if (!connection.ready) fail("SESSION_NOT_READY", "El WhatsApp de Julio no est\xE1 conectado en LR.", 503);
      }
      const image = await this.loadImage();
      if (digest(image) !== job.image_sha256) fail("IMAGE_CHANGED", "La imagen cambi\xF3. Es necesario revisar la versi\xF3n antes de enviar.");
      await this.ensureEligible(job);
      mayHaveSent = !dryRun;
      const result = await this.provider.submitImage({ sendId: job.id, to: job.recipient, caption: job.caption, image, mime: "image/jpeg", externalContactId: job.contact_id || job.company_id, consent: job.consent, dryRun });
      if (dryRun) {
        job = await this.store.updateJob(job, { status: "validated", lease_until: null, error_code: null, error_message: null });
        return publicJob(job);
      }
      job = await this.applyProvider(job, result);
      if (job.status === "sent") {
        try {
          job = await this.logSent(job);
        } catch {
          job = { ...job, log_pending: true };
        }
      }
      return publicJob(job);
    } catch (error) {
      const uncertain = mayHaveSent && !(error instanceof LrTransportError && !error.mayHaveBeenAccepted) && !(error instanceof LrApiError && error.status >= 400 && error.status < 500);
      const code = error instanceof OutboundError ? error.code : error instanceof LrApiError && SAFE_LR_CODES.has(error.code) ? error.code : uncertain ? "ACCEPTANCE_UNCERTAIN" : "CONNECTION_FAILED";
      const message = uncertain ? "No se pudo confirmar el resultado. Revisa este mismo intento; no crees otro env\xEDo." : error instanceof OutboundError ? error.message : "LR no acept\xF3 el env\xEDo. Revisa la conexi\xF3n o la configuraci\xF3n antes de volver a intentar.";
      let retryAt = null;
      if (error instanceof LrApiError && error.status === 429) {
        const raw = String(error.retryAfter || "");
        const time = /^\d+$/.test(raw) ? this.now().getTime() + Number(raw) * 1e3 : Date.parse(raw);
        retryAt = new Date(Number.isFinite(time) && time > this.now().getTime() && time < this.now().getTime() + 2592e6 ? time : this.now().getTime() + 6e4).toISOString();
      }
      job = await this.store.updateJob(job, { status: uncertain ? "uncertain" : "prepared", lease_until: retryAt, error_code: retryAt ? code === "QUEUE_LIMIT" ? "QUEUE_LIMIT" : "RATE_LIMITED" : code, error_message: retryAt ? "LR alcanz\xF3 su l\xEDmite de env\xEDo. Espera el plazo indicado antes de volver a intentar." : message });
      return publicJob(job);
    }
  }
  async applyProvider(job, result) {
    if (!result || !UUID2.test(result.messageId || "") || !LR_STATES.includes(result.status) || result.senderPhone && result.senderPhone !== "+51936755465") fail("INVALID_PROVIDER_RESPONSE", "LR devolvi\xF3 un estado que no se pudo comprobar.", 502);
    const reported = result.updatedAt;
    const time = typeof reported === "string" && Number.isFinite(Date.parse(reported)) && Date.parse(reported) <= this.now().getTime() + 3e5 ? new Date(reported).toISOString() : this.now().toISOString();
    return this.store.updateJob(job, { status: result.status, provider_message_id: result.messageId, provider_whatsapp_id: typeof result.whatsappMessageId === "string" ? result.whatsappMessageId.slice(0, 200) : null, sent_at: result.status === "sent" ? job.sent_at || time : null, lease_until: null, error_code: result.status === "failed" ? "LR_SEND_FAILED" : null, error_message: result.status === "failed" ? "LR inform\xF3 que no se pudo enviar. Revisa el intento antes de repetirlo." : null });
  }
  async logSent(job) {
    if (job.logged_at) return job;
    const activityId = "wa-" + job.id;
    for (let attempt = 0; attempt < 2; attempt++) {
      const data = await this.store.loadData();
      if (data.activities.some((row) => row.id === activityId)) return this.store.updateJob(job, { logged_at: this.now().toISOString(), error_code: null, error_message: null });
      const company = data.companies.find((row) => row.id === job.company_id);
      if (!company) return job;
      const now = new Date(job.sent_at || this.now());
      const operations = companyStage(company, data) === "perdida" || company.archived ? [{ type: "activities", expected: 0, record: { id: activityId, company_id: company.id, deal_id: "", kind: "contacto", channels: ["whatsapp"], outcome: "sin_respuesta", body: job.caption, occurred_at: now.toISOString(), date: today(now), time: localTime(now) } }] : contactOperations(data, company, ["whatsapp"], { body: job.caption, now });
      const record = operations.find((op2) => op2.type === "activities" && op2.record.kind === "contacto").record;
      record.id = activityId;
      record.outbound_id = job.id;
      record.provider_message_id = job.provider_message_id;
      try {
        await this.store.commit(operations);
        return await this.store.updateJob(job, { logged_at: this.now().toISOString(), error_code: null, error_message: null });
      } catch {
      }
    }
    return { ...job, log_pending: true };
  }
  async refreshInternal(job) {
    if (job.status === "sent") {
      try {
        return await this.logSent(job);
      } catch {
        return { ...job, log_pending: true };
      }
    }
    if (["submitting", "validating"].includes(job.status) && job.lease_until && Date.parse(job.lease_until) <= this.now().getTime()) return this.store.updateJob(job, { status: job.status === "submitting" ? "uncertain" : "prepared", lease_until: null, error_code: "INTERRUPTED", error_message: job.status === "submitting" ? "El intento se interrumpi\xF3. Puede haberse aceptado; revisa este mismo intento." : "La validaci\xF3n se interrumpi\xF3 sin solicitar env\xEDo." });
    if (!job.provider_message_id || !["queued", "processing", "uncertain"].includes(job.status)) return job;
    if (!this.configured) return job;
    try {
      job = await this.applyProvider(job, await this.provider.status(job.provider_message_id));
    } catch {
      return { ...job, refresh_pending: true };
    }
    if (job.status === "sent") {
      try {
        return await this.logSent(job);
      } catch {
        return { ...job, log_pending: true };
      }
    }
    return job;
  }
  async refresh(id2) {
    if (!UUID2.test(id2 || "")) fail("INVALID_REQUEST", "Identificador de env\xEDo inv\xE1lido.");
    return publicJob(await this.refreshInternal(await this.store.getJob(id2)));
  }
  async recover(id2) {
    if (!UUID2.test(id2 || "")) fail("INVALID_REQUEST", "Identificador de env\xEDo inv\xE1lido.");
    let job = await this.store.getJob(id2);
    job = await this.refreshInternal(job);
    if (job.provider_message_id || job.status !== "uncertain") return publicJob(job);
    return this.dispatch(job, false, true);
  }
  async cancel(id2) {
    if (!UUID2.test(id2 || "")) fail("INVALID_REQUEST", "Identificador de env\xEDo inv\xE1lido.");
    const job = await this.store.getJob(id2);
    if (!["prepared", "validated"].includes(job.status)) fail("JOB_BUSY", "Solo se puede descartar un borrador que todav\xEDa no se ha solicitado.");
    return publicJob(await this.store.updateJob(job, { status: "cancelled", lease_until: null }));
  }
};

async function authenticateSupabase(authorization, { service, makeUser }) {
  if (!/^Bearer \S+$/i.test(authorization || "")) throw new OutboundError("UNAUTHENTICATED", "Inicia sesi\xF3n para acceder a los env\xEDos.", 401);
  const { data, error } = await service.auth.getUser(authorization.replace(/^Bearer /i, ""));
  if (error || !data?.user) throw new OutboundError("UNAUTHENTICATED", "La sesi\xF3n no es v\xE1lida. Vuelve a iniciar sesi\xF3n.", 401);
  const user = makeUser(authorization);
  const { data: member, error: membershipError } = await user.from("crm_members").select("user_id").eq("user_id", data.user.id).maybeSingle();
  if (membershipError || !member) throw new OutboundError("NOT_AUTHORIZED", "Esta cuenta no est\xE1 autorizada en Nexo CRM.", 403);
  return { service, user, owner: data.user.id };
}

      const interrupted = jobs.find((j) => j.status === "sending" && Date.parse(j.lease_until || "") <= now.getTime());
      if (interrupted) {
        const logged = data.activities.find((a) => a.outbound_id === interrupted.send_id || a.id === "mail-" + interrupted.send_id || a.id === "wa-" + interrupted.send_id);
        if (logged) {
          await this.store.put({ ...interrupted, status: "sent", sent_at: logged.occurred_at || logged.created_at || now.toISOString(), error: "" });
          return { changed: true };
        }
        if (interrupted.channel === "whatsapp") try {
          const result = await this.whatsappFor(interrupted.template).refresh(interrupted.send_id);
          if (["sent", "queued", "processing", "uncertain", "failed"].includes(result.status)) {
            await this.store.put({ ...interrupted, status: result.status, provider_id: result.id, last_poll_at: now.toISOString(), sent_at: result.sent_at || "", error: result.error_message || "" });
            return { changed: true };
          }
        } catch {
        }
        await this.store.put({ ...interrupted, status: "uncertain", error: "El intento se interrumpi\xF3. Revisa el proveedor antes de repetirlo." });
        return { changed: true };
      }
      const awaiting = jobs.find((j) => ["queued", "processing"].includes(j.status) && (!j.last_poll_at || now.getTime() - Date.parse(j.last_poll_at) > 15e3));
      if (awaiting) {
        try {
          const api = this.whatsappFor(awaiting.template), result = await api.refresh(awaiting.provider_id);
          await this.store.put({ ...awaiting, status: result.status, provider_id: result.id, last_poll_at: now.toISOString(), sent_at: result.sent_at || "", error: result.error_message || "" });
        } catch {
        }
        return { changed: true };
      }
      const active = campaigns.filter((c) => c.status === "queued").sort((a, b) => String(a.created_at).localeCompare(String(b.created_at)) || a.id.localeCompare(b.id));
      for (const campaign of active) {
        const items = jobs.filter((j) => j.campaign_id === campaign.id).sort((a, b) => a.rank - b.rank);
        if (items.every((j) => terminal.includes(j.status))) {
          await this.store.put({ ...campaign, status: "completed" });
          continue;
        }
        const job = items.find((j) => j.status === "pending");
        if (!job) continue;
        const company = data.companies.find((c) => c.id === job.company_id);
        if (!company || company.archived || ["no_contactar", "no_encaja"].includes(company.status) || companyStage(company, data) === "perdida") {
          await this.store.put({ ...job, status: "skipped", error: "Empresa cerrada o archivada desde que se cre\xF3 la campa\xF1a." });
          return { changed: true };
        }
        if (job.channel === "whatsapp" && campaign.channel === "ambos") {
          const email = items.find((j) => j.company_id === job.company_id && j.channel === "correo");
          if (email && email.status !== "sent") {
            if (terminal.includes(email.status)) {
              await this.store.put({ ...job, status: "skipped", error: "El correo previo no se confirm\xF3; el mensaje lo menciona." });
              return { changed: true };
            }
            return { waiting: true };
          }
        }
        const key = job.channel === "correo" ? "next_email_at" : "next_whatsapp_at";
        if (Date.parse(lock[key] || "") > now.getTime()) return { nextAt: lock[key] };
        let claimed = await this.store.put({ ...job, status: "sending", lease_until: new Date(now.getTime() + 18e4).toISOString() });
        const interval = job.channel === "correo" ? campaign.email_interval : campaign.whatsapp_interval;
        schedule[key] = new Date(now.getTime() + interval * 1e3).toISOString();
        try {
          if (job.channel === "correo") {
            if (![company.email, ...data.contacts.filter((c) => c.company_id === company.id).map((c) => c.email)].some((email) => email && email.toLowerCase() === job.recipient.toLowerCase())) fail3("El correo cambi\xF3; revisa la empresa antes de enviar.", "RECIPIENT_CHANGED");
            const result = await this.emailApi.send({ sendId: job.send_id, companyId: job.company_id, toAddress: job.recipient, template: job.template, dryRun: false });
            if (result.status !== "sent") throw new OutboundError("EMAIL_UNCERTAIN", "Zoho no confirm\xF3 el resultado.", 503);
            await this.store.put({ ...claimed, status: "sent", sent_at: this.now().toISOString(), provider_id: result.providerId || "", error: "" });
          } else {
            const choices = recipientChoices(data, company), choice = choices.find((c) => (c.id || "") === job.contact_id);
            if (choice?.phone !== job.recipient) fail3("El tel\xE9fono cambi\xF3; revisa la empresa antes de enviar.", "RECIPIENT_CHANGED");
            const result = await this.whatsappFor(job.template).submit({ requestId: job.send_id, companyId: job.company_id, contactId: job.contact_id, caption: job.caption, emailAlreadySent: campaign.channel === "ambos" && items.some((j) => j.company_id === job.company_id && j.channel === "correo" && j.status === "sent"), consent: { confirmed: true, source: "Confirmaci\xF3n del operador al crear campa\xF1a", capturedAt: now.toISOString() }, dryRun: false });
            if (result.id !== job.send_id) {
              await this.store.put({ ...claimed, status: "skipped", provider_id: result.id, error: "Este mismo WhatsApp ya estaba registrado en LR; no se solicit\xF3 otro env\xEDo." });
              return { changed: true };
            }
            await this.store.put({ ...claimed, status: ["prepared", "validated"].includes(result.status) ? "failed" : result.status, provider_id: result.id, last_poll_at: now.toISOString(), sent_at: result.sent_at || "", error: result.error_message || "" });
            if (result.retry_at) {
              schedule[key] = result.retry_at;
              await this.store.put({ ...campaign, status: "paused", error: "El proveedor pidi\xF3 esperar. Revisa el plazo del intento antes de continuar." });
            } else if (["prepared", "validated", "uncertain", "failed"].includes(result.status)) {
              await this.store.put({ ...campaign, status: "paused", error: "WhatsApp no confirm\xF3 el env\xEDo. Revisa el intento antes de continuar." });
            }
          }
        } catch (error) {
          const definite = ["SENDING_DISABLED", "EMAIL_REQUIRED", "COMPANY_NOT_FOUND", "CONTACT_BLOCKED", "RECIPIENT_CHANGED", "ZOHO_NOT_CONFIGURED", "ZOHO_AUTH_FAILED", "ZOHO_ACCOUNT_FAILED", "ZOHO_SEND_REJECTED", "RATE_LIMITED", "SENDER_NOT_AUTHORIZED", "SENDER_OWNER_NOT_CONFIGURED", "LR_NOT_CONFIGURED"].includes(error.code);
          try {
            await this.store.put({ ...claimed, status: definite ? "failed" : "uncertain", error: error instanceof OutboundError ? error.message : "No se pudo confirmar el env\xEDo. Revisa el proveedor antes de repetirlo." });
          } catch {
          }
          await this.store.put({ ...campaign, status: "paused", error: "La campa\xF1a se paus\xF3 por un error. El historial conserva el intento." });
          if (error.code === "RATE_LIMITED") schedule[key] = new Date(now.getTime() + 3e5).toISOString();
        }
        return { changed: true, version: TEMPLATE_VERSION };
      }
      return { idle: true, version: TEMPLATE_VERSION };
    } finally {
      await this.store.release(lock, schedule);
    }
  }
};
