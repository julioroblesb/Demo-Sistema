function contactHistory(companyId, data) {
  const rows = (data.activities || []).filter((row) => row.company_id === companyId);
  const attempts = rows.filter((row) => row.kind === "contacto" || ["correo", "whatsapp", "llamada", "reunion", "presencial"].includes(row.kind));
  attempts.sort((a, b) => activityTime(b).localeCompare(activityTime(a)));
  const contact = attempts[0];
  if (!contact) {
    const responses2 = rows.filter((row) => row.kind === "respuesta").sort((a, b) => activityTime(b).localeCompare(activityTime(a)));
    return { contact: null, at: "", channels: [], response: responses2[0] || null };
  }
  const at = activityTime(contact);
  const group = attempts.filter((row) => activityTime(row) === at);
  const channels = [...new Set(group.flatMap((row) => row.kind === "contacto" ? row.channels || [] : [row.kind === "reunion" ? "presencial" : row.kind]))].filter((key) => CONTACT_CHANNELS[key]);
  const responses = rows.filter((row) => row.kind === "respuesta" && (row.contact_id === contact.id || !row.contact_id && activityTime(row) >= at));
  responses.push(...group.filter((row) => ["conversacion", "no_interes", "no_contactar"].includes(row.outcome)));
  responses.sort((a, b) => activityTime(b).localeCompare(activityTime(a)));
  return { contact, at, channels, response: responses[0] || null };
}
function companyStage(company, data) {
  if (STAGES[company.pipeline_stage]) return company.pipeline_stage;
  if (["no_contactar", "no_encaja"].includes(company.status)) return "perdida";
  const deals = (data.deals || []).filter((row) => row.company_id === company.id && !row.archived);
  const open = deals.filter(isOpen);
  if (open.some((row) => dealStageKey(row) === "propuesta")) return "propuesta";
  if (!open.length && (deals.some((row) => row.stage === "ganada") || company.status === "cliente")) return "ganada";
  if (!open.length && deals.some((row) => row.stage === "perdida")) return "perdida";
  if (["conversacion", "calificada"].includes(company.status)) return "propuesta";
  const history = contactHistory(company.id, data);
  if (history.response?.outcome === "conversacion") return "propuesta";
  if (history.response && ["no_interes", "no_contactar"].includes(history.response.outcome)) return "perdida";
  if (history.contact || company.status === "contactada" || open.some((row) => dealStageKey(row) === "demo")) return "demo";
  return "diagnostico";
}

function stageOperations(data, company, stage, { reason = "", now = /* @__PURE__ */ new Date(), allowContact = false } = {}) {

function contactOperations(data, company, channels, { body = "", move = false, allowReopen = false, now = /* @__PURE__ */ new Date() } = {}) {

  if (current === "perdida" && !allowReopen) throw new Error("El seguimiento est\xE1 cerrado. Reabre la empresa antes de contactarla.");

  let operations;
  if (stage !== current || move) operations = stageOperations(data, company, stage, { allowContact: true, now });

  return assertSize(operations);

function normalizeWhatsAppPhone(value) {
  let digits = String(value || "").replace(/\D/g, "").replace(/^00/, "");
  if (/^9\d{8}$/.test(digits)) digits = "51" + digits;

}

    if (job.status === "sent") {
      try {
        return await this.logSent(job);
      } catch {
        return { ...job, log_pending: true };
      }
    }

    if (!this.configured) return job;
    try {
      job = await this.applyProvider(job, await this.provider.status(job.provider_message_id));
    } catch {
      return { ...job, refresh_pending: true };
    }
    if (job.status === "sent") {
      try {
        return await this.logSent(job);
      } catch {
        return { ...job, log_pending: true };
      }
    }
    return job;
  }
  async refresh(id2) {
    if (!UUID2.test(id2 || "")) fail("INVALID_REQUEST", "Identificador de env\xEDo inv\xE1lido.");

    let job = await this.store.getJob(id2);
    job = await this.refreshInternal(job);
    if (job.provider_message_id || job.status !== "uncertain") return publicJob(job);
    return this.dispatch(job, false, true);
  }
  async cancel(id2) {
    if (!UUID2.test(id2 || "")) fail("INVALID_REQUEST", "Identificador de env\xEDo inv\xE1lido.");
    const job = await this.store.getJob(id2);
    if (!["prepared", "validated"].includes(job.status)) fail("JOB_BUSY", "Solo se puede descartar un borrador que todav\xEDa no se ha solicitado.");
    return publicJob(await this.store.updateJob(job, { status: "cancelled", lease_until: null }));
  }
};

function createWhatsAppHandler({ authenticate, makeService, origin = "https://nexoamedida.com" }) {
  const cors = { "Access-Control-Allow-Origin": origin, "Access-Control-Allow-Headers": "authorization,apikey,content-type,x-client-info", "Access-Control-Allow-Methods": "POST,OPTIONS", "Cache-Control": "no-store", "Vary": "Origin" };
  const reply = (body, status = 200) => new Response(JSON.stringify(body), { status, headers: { ...cors, "Content-Type": "application/json; charset=utf-8" } });
  return async (req) => {
    if (req.headers.get("origin") && req.headers.get("origin") !== origin) return reply({ error: { code: "ORIGIN_DENIED", message: "Origen no autorizado." } }, 403);
    if (req.method === "OPTIONS") return new Response(null, { status: 204, headers: cors });
    if (req.method !== "POST") return reply({ error: { code: "METHOD_NOT_ALLOWED", message: "Usa POST." } }, 405);
    try {
      const authorization = req.headers.get("authorization") || "";
      if (!/^Bearer \S+$/i.test(authorization)) throw new OutboundError("UNAUTHENTICATED", "Inicia sesi\xF3n para acceder a los env\xEDos.", 401);
      const identity = await authenticate(authorization);
      if (Number(req.headers.get("content-length") || 0) > 12e3) throw new OutboundError("INVALID_REQUEST", "La solicitud es demasiado grande.", 413);
      const text = await req.text();
      if (text.length > 12e3) throw new OutboundError("INVALID_REQUEST", "La solicitud es demasiado grande.", 413);
      let input;
      try {
        input = JSON.parse(text);
      } catch {
        throw new OutboundError("INVALID_REQUEST", "La solicitud no es v\xE1lida.");
      }
      if (!input || typeof input !== "object" || Array.isArray(input)) throw new OutboundError("INVALID_REQUEST", "La solicitud no es v\xE1lida.");
      const api = await makeService(identity, input);
      if (input.action === "context") return reply(await api.context(input.companyId));
      if (input.action === "submit") return reply({ job: await api.submit(input) });
      if (input.action === "refresh") return reply({ job: await api.refresh(input.jobId) });
      if (input.action === "recover") return reply({ job: await api.recover(input.jobId) });
      if (input.action === "cancel") return reply({ job: await api.cancel(input.jobId) });
      throw new OutboundError("INVALID_REQUEST", "Acci\xF3n no disponible.");
    } catch (error) {
      const problem = error instanceof OutboundError ? error : new OutboundError("INTEGRATION_ERROR", "No se pudo completar la operaci\xF3n. Consulta el historial antes de repetirla.", 503);
      return reply({ error: { code: problem.code, message: problem.message } }, problem.status);
    }
  };
}

Deno.serve(createWhatsAppHandler({
  authenticate: async (authorization) => {
    const url = Deno.env.get("SUPABASE_URL"), serviceKey = Deno.env.get("SUPABASE_SERVICE_ROLE_KEY"), anonKey = Deno.env.get("SUPABASE_ANON_KEY");
    if (!url || !serviceKey || !anonKey) throw new OutboundError("BACKEND_NOT_CONFIGURED", "La funci\xF3n no tiene configurada su conexi\xF3n interna.", 503);
    const service = createClient(url, serviceKey, { auth: { persistSession: false, autoRefreshToken: false } });
    return authenticateSupabase(authorization, { service, makeUser: (header) => createClient(url, anonKey, { auth: { persistSession: false, autoRefreshToken: false }, global: { headers: { Authorization: header } } }) });
  },
  makeService: async (identity, input) => {
    let template = input.template;
    if (["refresh", "recover", "cancel"].includes(input.action)) {
      const { data } = await identity.service.from("crm_whatsapp_jobs").select("image_sha256").eq("owner_id", identity.owner).eq("id", input.jobId).maybeSingle();
      template = Object.entries(campaign_default.sectors || {}).find(([, asset]) => asset.sha256 === data?.image_sha256)?.[0] || "general";
    } else if (!template) {
      const { data } = await identity.user.from("crm_records").select("data").eq("kind", "companies").eq("id", input.companyId).maybeSingle();
      template = sectorFor(data?.data);
    }
    const api = whatsappRuntime(identity, template), context = api.context.bind(api);
    api.context = async (companyId) => ({ ...await context(companyId), templateVersion: TEMPLATE_VERSION, sector: template });
    return api;
  }
}));
